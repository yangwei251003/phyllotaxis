cmake_minimum_required(VERSION 3.20)
project(phyllo_portable_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(phyllo STATIC src/phyllo.c)
target_include_directories(phyllo PUBLIC include)
target_compile_options(phyllo PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE phyllo m)
target_compile_options(test_core PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
enable_testing()
add_test(NAME portable_signal_and_led_contract COMMAND test_core)
